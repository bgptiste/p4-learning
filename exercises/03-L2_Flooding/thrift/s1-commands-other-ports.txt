# 1 groupe multicast par port de sortie (1,2,3,4): 
mc_mgrp_create 1
mc_mgrp_create 2
mc_mgrp_create 3
mc_mgrp_create 4

# Création des noeuds multicast :
mc_node_create 0 2 3 4
mc_node_create 1 1 3 4
mc_node_create 2 1 2 4
mc_node_create 3 1 2 3

# Association des noeuds multicast au groupe :
mc_node_associate 1 0
mc_node_associate 2 1
mc_node_associate 3 2
mc_node_associate 4 3

# Association adresse MAC vers port :
table_add forwarding setEgress 00:00:0a:00:00:01 => 1
table_add forwarding setEgress 00:00:0a:00:00:02 => 2
table_add forwarding setEgress 00:00:0a:00:00:03 => 3
table_add forwarding setEgress 00:00:0a:00:00:04 => 4

# Association port vers groupe multicast (1,2,3,4) :
table_add flooding setMcastGrp 1 => 1
table_add flooding setMcastGrp 2 => 2
table_add flooding setMcastGrp 3 => 3
table_add flooding setMcastGrp 4 => 4
